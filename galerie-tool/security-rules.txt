// Firestore Security Rules
// In der Firebase Console unter "Firestore Database" -> "Regeln" einfügen

rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    match /galerien/{galerieId} {
      // Jeder darf eine Galerie lesen (nötig, damit Kunden sie öffnen können)
      // ABER: passwortHash wird zwar mitgelesen, ist aber ein SHA-256 Hash, kein Klartext
      allow read: if true;

      // Nur eingeloggte Admins (also du) dürfen Galerien erstellen/ändern
      allow write: if request.auth != null;

      match /bilder/{bildId} {
        allow read: if true;

        // Erstellen/Löschen nur für dich (Admin)
        allow create, delete: if request.auth != null;

        // ABER: Das Favorit-Feld dürfen auch Kunden ohne Login ändern
        // (das ist eine bewusste Vereinfachung für dieses Tool)
        allow update: if request.auth != null ||
                         (request.resource.data.diff(resource.data).affectedKeys().hasOnly(['favorit']));
      }
    }
  }
}

// Storage Security Rules
// In der Firebase Console unter "Storage" -> "Regeln" einfügen

rules_version = '2';
service firebase.storage {
  match /b/{bucket}/o {
    match /galerien/{galerieId}/preview/{datei} {
      allow read: if true;  // Vorschaubilder mit Wasserzeichen sind öffentlich sichtbar
      allow write: if request.auth != null;
    }
    match /galerien/{galerieId}/original/{datei} {
      // Original-Dateien NUR für eingeloggte Admins lesbar
      // (Download-Freigabe für Kunden würde eine Cloud Function brauchen -> spätere Ausbaustufe)
      allow read: if request.auth != null;
      allow write: if request.auth != null;
    }
  }
}
